def get(url):
 with urllib.request.urlopen(url,timeout=5) as r:return json.load(r)
def idle_check():
 running=get('http://127.0.0.1:8080/running')['running']
 for m in running:
  if m.get('state')!='ready':raise RuntimeError('Model is not idle/ready: '+m.get('state','?'))
  try:
   slots=get(m['proxy']+'/slots')
   if any(s.get('is_processing',False) for s in slots):raise RuntimeError('Active model request detected; will not interrupt')
  except urllib.error.HTTPError as e:
   if e.code in [404,501]:
    h=get(m['proxy']+'/health?include_slots=true')
    if 'slots_processing' not in h:raise RuntimeError('Cannot verify idle slots')
    if h['slots_processing']:raise RuntimeError('Active slots detected')
   else:raise
 return [m['model'] for m in running]
